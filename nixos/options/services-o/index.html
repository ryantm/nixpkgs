services.oxidized.settings

Oxidized configuration as a Nix attribute set installed at /etc/oxidized/config.

The resulting configuration file resides in the /nix/store. Pass secret values via services.oxidized.userConfigFile instead.

This option is mutually exclusive with services.oxidied.configFile.

Note

Ruby-specific YAML tags can be passed as a quoted string. The config renderer will unquote tags starting with !ruby/ in both keys and values.

Type: null or YAML 1.1 value

Default: null

Example:

{
  debug = true;
  extensions = {
    oxidized-web = {
      listen = "127.0.0.1";
      load = true;
      port = 8888;
      vhosts = [
        "localhost"
        "127.0.0.1"
        "oxidized.example.com"
      ];
    };
  };
  input = {
    default = "ssh";
    ssh = {
      secure = true;
    };
  };
  interval = 3600;
  model_map = {
    cisco = "ios";
    dell = "powerconnect";
    hp = "procurve";
    juniper = "junos";
  };
  pid = "/var/lib/oxidized/.config/oxidized/pid";
  retries = 3;
  source = {
    csv = {
      delimiter = "!ruby/regexp /:/";
      file = "/var/lib/oxidized/.config/oxidized/router.db";
      map = {
        model = 1;
        name = 0;
        password = 3;
        username = 2;
      };
    };
    default = "csv";
  };
  use_syslog = true;
}
